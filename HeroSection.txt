<script>
        document.addEventListener('DOMContentLoaded', () => {
            const bgImage = document.getElementById('hero-bg');
            const scrollContent = document.getElementById('scroll-content');
            const trustContent = document.getElementById('trust-content');
            
            let ticking = false;

            function updateScrollEffect() {
                const scrollY = window.scrollY;
                const maxScroll = 800; // Define the scroll range for the effect
                
                // Calculate progress (0 to 1)
                const progress = Math.min(scrollY / maxScroll, 1);
                
                // Scale from 1 to 1.3
                const scale = 1 + (progress * 0.3);
                
                // Opacity from 1 to 0
                const opacity = 1 - progress;
                
                // Apply transforms using requestAnimationFrame for smoothness
                bgImage.style.transform = `scale(${scale})`;
                
                scrollContent.style.opacity = opacity;
                scrollContent.style.pointerEvents = opacity > 0 ? 'auto' : 'none';
                
                trustContent.style.opacity = opacity;

                ticking = false;
            }

            window.addEventListener('scroll', () => {
                if (!ticking) {
                    window.requestAnimationFrame(updateScrollEffect);
                    ticking = true;
                }
            });

            // Initial call to set correct state on load
            updateScrollEffect();
        });
    </script>
