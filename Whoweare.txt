            cards.forEach(card => {
                card.addEventListener('mouseenter', () => {
                    const targetId = card.getAttribute('data-target');
                    
                    // Reset all callouts
                    callouts.forEach(c => c.classList.remove('active'));
                    
                    // Activate target callout
                    const targetCallout = document.getElementById(targetId);
                    if(targetCallout) {
                        targetCallout.classList.add('active');
                    }
                });
            });
